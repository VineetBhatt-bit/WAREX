cmake_minimum_required(VERSION 3.20)
project(WAREX VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WAREX_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(warex_ds STATIC src/c/warehouse_ds.c)
target_include_directories(warex_ds PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(warex_ds PRIVATE -Wall -Wextra -Wpedantic)

add_library(warex_service STATIC src/cpp/warehouse_service.cpp)
target_include_directories(warex_service PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(warex_service PUBLIC warex_ds)
target_compile_options(warex_service PRIVATE -Wall -Wextra -Wpedantic)

add_executable(warex src/cpp/http_server.cpp)
target_link_libraries(warex PRIVATE warex_service)
target_compile_definitions(warex PRIVATE WAREX_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(warex PRIVATE -Wall -Wextra -Wpedantic)

if(WAREX_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  foreach(target warex_ds warex_service warex)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(warex_tests tests/test_warehouse.cpp)
  target_link_libraries(warex_tests PRIVATE warex_service)
  target_compile_options(warex_tests PRIVATE -Wall -Wextra -Wpedantic)
  if(WAREX_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(warex_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(warex_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME warex_tests COMMAND warex_tests)
endif()
